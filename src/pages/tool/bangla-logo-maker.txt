    <script is:inline>
      const logoPreview = document.getElementById('logoPreview');
      const bgColorInput = document.getElementById('bgColor');
      const bgImageInput = document.getElementById('bgImage');
      const downloadBtn = document.getElementById('downloadBtn');

      // Font selection
      document.querySelectorAll('.font-option').forEach(opt => {
        opt.addEventListener('click', () => {
          document.querySelectorAll('.font-option').forEach(o => o.classList.remove('selected'));
          opt.classList.add('selected');
          logoPreview.style.fontFamily = opt.dataset.fontfamily;
        });
      });

      // Background color exclusive
      bgColorInput.addEventListener('input', () => {
        logoPreview.style.backgroundColor = bgColorInput.value;
        logoPreview.style.backgroundImage = '';
        bgImageInput.value = '';
      });

      // Background image exclusive
      bgImageInput.addEventListener('change', (e) => {
        const file = e.target.files[0];
        if (file && file.type.startsWith('image/')) {
          const reader = new FileReader();
          reader.onload = (event) => {
            // ✅ Fixed string concatenation instead of backticks
            logoPreview.style.backgroundImage = "url('" + event.target.result + "')";
            logoPreview.style.backgroundSize = 'cover';
            logoPreview.style.backgroundPosition = 'center';
            logoPreview.style.backgroundColor = '';
            bgColorInput.value = '#ffffff';
          };
          reader.readAsDataURL(file);
        }
      });

      // Download logo as image
      downloadBtn.addEventListener('click', () => {
        html2canvas(logoPreview).then(canvas => {
          const link = document.createElement('a');
          link.download = 'logo.jpeg';
          link.href = canvas.toDataURL('image/jpeg', 1.0);
          link.click();
        });
      });
    </script>
